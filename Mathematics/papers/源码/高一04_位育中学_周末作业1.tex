\bigsec{一、填空题}

\begin{enumerate}
\setcounter{enumi}{2}

\item 用列举法表示集合 $A=\left\{y\,\middle|\,y=\dfrac{12}{3-x},\,x\in\mathbb{N},\,x\in\mathbb{Z}\right\}=$ \blankline[2em].

\ans{$\{1,2,3,4,6,12\}$}

\item 已知集合 $A=\{2a,\,2a^2+a\}$，若 $1\in A$，则 $a=$ \blankline[2em].

\ans{$-1$.}

\ifanswer
\textbf{【解析】}因为 $1\in A$，所以 $2a=1$ 或 $2a^2+a=1$. 当 $2a=1$ 时，$a=\tfrac{1}{2}$，此时 $2a^2+a=1$，不满足集合中元素的互异性，舍去；当 $2a^2+a=1$ 时，解得 $a=-1$ 或 $a=\tfrac{1}{2}$（舍去），此时 $A=\{-2,1\}$. 综上，$a=-1$.

\fi
\item 关于 $x,y$ 的方程组 $\begin{cases} x+y=2 \\ x^2+y^2=2 \end{cases}$ 的解集为 \blankline[2em].

\ans{$\{(1,1)\}$}

\item 如果集合 $A=\{x\mid y=x^2,\,x\in\R\}$，$B=\{y\mid y=x^2,\,x\in\R\}$，则集合 $A$ 与 $B$ 的关系是 \blankline[2em].

\ans{$B\subsetneq A$}

\item 已知 $A=\left\{x\,\middle|\,x=\dfrac{k}{3}-\dfrac{1}{2},\,k\in\Z\right\}$，$B=\left\{x\,\middle|\,x=\dfrac{k}{6}+\dfrac{2}{3},\,k\in\Z\right\}$，则集合 $A$ 与 $B$ 的关系是 \blankline[2em].

\ans{$A\supsetneq B$（或 $A=B$ 之等价条件）}

\item 若非空集合 $\{x\mid m\leqslant x\leqslant 2m-\sqrt{m}\}$ 不存在非空真子集，则 $m$ 的取值集合为 \blankline[2em].

\ans{$\{0,1\}$}

\item 已知 $P=\{x\mid 2<x<a,\,x\in\mathbb{N}\}$，且集合 $P$ 中恰有 3 个元素，则实数 $a$ 的取值范围为 \blankline[3em]（用区间表示）.

\ans{$(5,6]$}

\item 设 $x_1,x_2,x_3,x_4$ 是 4 个正整数，从中任取 3 个数求和所得的集合为 $\{25,26,27\}$，则这 4 个数中最小的数为 \blankline[2em].

\ans{$8$}

\ifanswer
\textbf{【解析】}从 4 个正整数中任取 3 个数求和后可得到 4 个和，则 4 个和值之和为 $3(x_1+x_2+x_3+x_4)$，必为 3 的倍数. 又 $27\div 3=9,\,25\div 3=8\dots1,\,26\div 3=8\dots2$，所以这 4 个和为 $25,26,27,27$. 则 $x_1+x_2+x_3+x_4=\tfrac{1}{3}(25+26+27+27)=35$. 所以 $35-25=10,\,35-26=9,\,35-27=8$，即这 4 个数分别为 $8,8,9,10$，故这 4 个数中最小的数为 8.

\fi
\end{enumerate}

\bigsec{二、选择题}

\begin{enumerate}
\setcounter{enumi}{10}

\item 下列说法中正确的是（\quad）.

\ans{B}

A. $0$ 与 $\{0\}$ 表示同一个集合；

B. 集合 $\{1,2,3\}$ 与 $\{3,2,1\}$ 是两个相同的集合；

C. 方程 $(x-1)^2(x-2)=0$ 的解集为 $\{1,1,2\}$；

D. 集合 $\{x\mid 4<x<5\}$ 可以用列举法表示.

\item 下列说法正确的是（\quad）.

\ans{C}

A. $\varnothing\in\{0\}$\qquad B. $0\subseteq\{0\}$\qquad C. $\varnothing\subseteq\{0,1,2\}$\qquad D. $(1,2)\supset\{1,2\}$

\item 已知非空集合 $A=\{x\mid x^2-2x+m=0\}$，则集合 $A$ 中所有元素之和为（\quad）.

\ans{C}

A. $1$\qquad B. $2$\qquad C. $1$ 或 $2$\qquad D. 非以上答案

\item 如果集合 $S$ 具有性质：① 非空且它的元素都是正整数；② 如果 $x\in S$，那么 $\tfrac{24}{x}\in S$. 请问这样的集合 $S$ 共有（\quad）个.

\ans{A}

A. $15$ 个\qquad B. $16$ 个\qquad C. $31$ 个\qquad D. $32$ 个

\end{enumerate}
