\bigsec{一、填空题}

\begin{enumerate}
\setcounter{enumi}{2}

\item 若 $-1<x<y<2$，则 $x-y$ 的取值范围为 \blankline[4.4em].

\ifanswer
\solbegin
\step{$(-3,0)$.}
\solend
\fi

\item 已知 $c>d$，若 $ac+bd>bc+ad$，则 $a-b$ 与 $0$ 的大小关系是 \blankline[4.4em].

\ifanswer
\solbegin
\step{$a-b>0$.}
\solend
\fi

\item 已知 $a>b>0$，$P=\sqrt[3]{a}-\sqrt[3]{b}$，$Q=\sqrt[3]{a-b}$，则 $P$、$Q$ 的大小关系是 \blankline[4.4em].

\ifanswer
\solbegin
\step{$P>Q$.}
\solend
\fi

\item 对任意的实数 $x$，不等式 $x+|x-1|>m$ 恒成立，则实数 $m$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
\step{令 $f(x)=x+|x-1|=\begin{cases}2x-1,&x\geqslant1,\\1,&x<1.\end{cases}$，所以 $(x+|x-1|)_{\min}=1$，所以 $m<1$.}
\solend
\fi

\item 若“存在 $x\in\{x\mid1\leqslant x\leqslant2\}$，使得 $3x+a+1\geqslant0$”是真命题，则 $a$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
\step{由题意得 $6+a+1\geqslant0$，所以 $a\geqslant-7$.}
\solend
\fi

\item 已知关于 $x$ 的不等式 $ax-b\leqslant0$ 的解集是 $[2,+\infty)$，则关于 $x$ 的不等式
$ax^2+(3a-b)x-3b<0$ 的解集是 \blankline[4.4em].

\ifanswer
\solbegin
\step{由题意得 $a<0$，$\dfrac ba=2$，所以 $b=2a$，}
\step{则 $ax^2+(3a-b)x-3b<0$ 即 $ax^2+ax-6a<0$，所以 $x^2+x-6>0$，}
\step{解集为 $(-\infty,-3)\cup(2,+\infty)$.}
\solend
\fi

\item 若关于 $x$ 的不等式 $(k^2+4k-5)x^2+4(1-k)x+3>0$ 的解集为 $\R$，则实数 $k$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
\step{当 $k^2+4k-5=0$ 时，$k=-5$ 或 $k=1$.}
\step{当 $k=-5$ 时，不等式变为 $24x+3>0$，显然不满足题意，所以 $k\neq-5$.}
\step{当 $k=1$ 时，不等式变为 $3>0$，这时 $x\in\R$.}
\step{当 $k^2+4k-5\neq0$，有 $\begin{cases}k^2+4k-5>0\\\Delta<0\end{cases}\Leftrightarrow1<k<19$，}
\step{综上，实数 $k$ 的取值范围为 $[1,19)$.}
\solend
\fi

\item 设 $a\in\R$，集合 $A=\{x\mid x^2-ax+a\leqslant0\}$，若 $A\cap\N\neq\varnothing$，则 $a$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
\step{$(-\infty,0]\cup[4,+\infty)$.}
\solend
\fi

\item 用 $|S|$ 表示集合 $S$ 中的元素的个数，设 $A$、$B$、$C$ 为集合，称 $(A,B,C)$ 为有序三元组。如果集合 $A$、$B$、$C$ 满足 $|A\cap B|=|B\cap C|=|C\cap A|=1$，且 $|A\cap B\cap C|=0$，则称有序三元组 $(A,B,C)$ 为最小相交。由集合 $\{1,2,3,4\}$ 的子集构成的所有有序三元组中，最小相交的有序三元组的个数为 \blankline[4.4em].

\ifanswer
\solbegin
\step{因为 $|A\cap B|=|B\cap C|=|C\cap A|=1$，}
\step{所以设 $A\cap B=\{x\}$，$B\cap C=\{y\}$，$C\cap A=\{z\}$，}
\begin{center}
\begin{tikzpicture}[scale=0.78]
\draw[gray] (0,0.75) circle (1.2);
\draw[gray] (-0.9,-0.55) circle (1.2);
\draw[gray] (0.9,-0.55) circle (1.2);
\node at (0,1.65) {$C$};
\node at (-1.8,-0.9) {$A$};
\node at (1.8,-0.9) {$B$};
\node at (-0.64,0.20) {$z$};
\node at (0.64,0.20) {$y$};
\node at (0,-0.88) {$x$};
\end{tikzpicture}
\end{center}
\step{因为 $A\cap B\cap C=\varnothing$，且 $x$、$y$、$z\in\{1,2,3,4\}$，}
\step{①集合 $\{1,2,3,4\}$ 中的子集含有 4 个元素时，}
\step{所以从 $1,2,3,4$ 四个元素选 3 个有 $C_4^3=4$ 种方法，}
\step{将 3 个元素进行全排列有 $P_3^3=3\times2=6$ 种，}
\step{剩余的一个元素可以分别放入集合 $A$、$B$、$C$，有 3 种，}
\step{所以此时共有 $3\times4\times6=72$ 种.}
\step{②集合 $\{1,2,3,4\}$ 中的子集含有 3 个元素时，}
\step{满足集合 $A$、$B$、$C$ 中都只有一个元素.}
\step{所以从 $1,2,3,4$ 四个元素选 3 个有 $C_4^3=4$ 种方法，}
\step{将 3 个元素进行全排列有 $P_3^3=3\times2=6$ 种，所以此时共有 $4\times6=24$ 种.}
\step{综上，共有 $72+24=96$ 个.}
\solend
\fi

\item 设 $a\in\R$，记 $f(x)=x^2+2x+a$，若关于 $x$ 的不等式 $f(f(x))<0$ 的解集为 $\varnothing$，则 $a$ 的取值范围为 \blankline[4.4em].

\ifanswer
\solbegin
\step{二次函数 $f(x)=x^2+2x+a$ 的开口向上，对称轴为 $x=-1$，}
\step{所以 $f(x)_{\min}=f(-1)=a-1$，即 $f(x)$ 的值域为 $[a-1,+\infty)$，}
\step{要使不等式 $f(f(x))<0$ 的解集为空集，则 $f(f(x))\geqslant0$ 恒成立，}
\step{所以 $\begin{cases}f(a-1)\geqslant0,\\a-1\geqslant-1,\end{cases}$ 即 $\begin{cases}a^2+a-1\geqslant0,\\a\geqslant0,\end{cases}$ 解得 $a\geqslant\dfrac{\sqrt5-1}{2}$，}
\step{所以实数 $a$ 的取值范围为 $\left[\dfrac{\sqrt5-1}{2},+\infty\right)$.}
\solend
\fi

\end{enumerate}

\bigsec{二、选择题}

\begin{enumerate}
\setcounter{enumi}{12}

\item 已知关于 $x$ 的不等式 $2x-1>a(x-2)$ 的解集是 $\R$，则实数 $a$ 的取值范围是\mbox{（\quad）}

\keepwithprev
A. $a>2$\qquad B. $a=2$

C. $a<2$\qquad D. $a$ 不存在

\ans{$B$}

\item 设 $0<a_1<a_2$，$0<b_1<b_2$，且 $a_1+a_2=b_1+b_2=1$，如果要把 $a_1b_1+a_2b_2$、$a_1b_2+a_2b_1$、$\dfrac12$ 按从小到大的顺序排列，那么，排在中间的数\mbox{（\quad）}

\keepwithprev
A. 一定是 $a_1b_1+a_2b_2$\qquad B. 一定是 $a_1b_2+a_2b_1$

C. 一定是 $\dfrac12$\qquad D. 不能确定，与 $a_1,a_2,b_1,b_2$ 的值有关

\ans{$C$}

\item “$0<a<b$”是“$a-\dfrac1a<b-\dfrac1b$”的\mbox{（\quad）}条件

\keepwithprev
A. 充分不必要\qquad B. 必要不充分

C. 充要\qquad D. 既不充分也不必要

\ans{$A$}

% 注：题干与选项中 f 的下标原卷印作小写 s（解析中用大写 S），原卷如此，照录未改。
\item 在 $n$ 元数集 $S=\{a_1,a_2,\cdots,a_n\}$ 中，设 $x(S)=\dfrac{a_1+a_2+\cdots+a_n}{n}$，若 $S$ 的非空子集 $A$ 满足 $x(A)=x(S)$，则称 $A$ 是集合 $S$ 的一个“平均子集”，并记数集 $S$ 的 $k$ 元“平均子集”的个数为 $f_s(k)$. 已知集合 $S=\{1,2,3,4,5,6,7,8,9\}$，$T=\{-4,-3,-2,-1,0,1,2,3,4\}$，则下列说法错误的是\mbox{（\quad）}

\keepwithprev
A. $f_s(9)=f_T(1)$\qquad B. $f_s(8)=f_T(1)$

C. $f_s(6)=f_T(4)$\qquad D. $f_s(5)=f_T(4)$

\ans{$D$}

\ifanswer
\solbegin
\step{$X(S)=5$，将 $S$ 中的元素分成 5 组 $(1,9)$、$(2,8)$、$(3,7)$、$(4,6)$、$(5)$.}
\step{则 $f_S(1)=C_1^1=1$，$f_S(2)=C_4^1=4$，$f_S(3)=C_1^1C_4^1=4$，$f_S(4)=C_4^2=6$，}
\step{$f_S(5)=C_1^1C_4^2=6$，同理 $X(T)=0$，}
\step{将 $T$ 中的元素分成 5 组 $(1,-1)$、$(2,-2)$、$(3,-3)$、$(4,-4)$、$(0)$.}
\step{则 $f_T(1)=C_1^1=1$，$f_T(2)=C_4^1=4$，$f_T(3)=C_1^1C_4^1=4$，$f_T(4)=C_4^2=6$，}
\step{$f_T(5)=C_1^1C_4^2=6$，$f_T(8)=C_4^4=1$，所以 $f_S(4)=f_S(5)=f_T(4)=6$. 故选 D.}
\solend
\fi

\end{enumerate}

\bigsec{三、解答题}

\begin{enumerate}
\setcounter{enumi}{16}

\item 已知集合 $A=\{x\mid x^2-12x+20\leqslant0\}$，$B=\{x\mid2m<x<3m+4\}$.

\begin{enumerate}[label=(\arabic*)]
\item 若 $m=3$，求 $A\cup\overline B$；
\item 若 $A\cap B=B$，求 $m$ 的取值范围.
\end{enumerate}

\ifanswer
\solbegin
\stepm{(1)}{因为 $A=\{x\mid x^2-12x+20\leqslant0\}=\{x\mid2\leqslant x\leqslant10\}$，}
\step{当 $m=3$ 时，$B=\{x\mid6<x<13\}$，则 $\overline B=\{x\mid x\leqslant6$ 或 $x\geqslant13\}$，}
\step{所以，$A\cup\overline B=\{x\mid x\leqslant10$ 或 $x\geqslant13\}$.}
\stepm{(2)}{因为 $A\cap B=B$，所以 $B\subseteq A$，}
\step{当 $B=\varnothing$ 时，$2m\geqslant3m+4$，解得 $m\leqslant-4$；}
\step{当 $B\neq\varnothing$ 时，由 $B\subseteq A$ 得 $\begin{cases}2m<3m+4\\2m\geqslant2\\3m+4\leqslant10\end{cases}$，解得 $1\leqslant m\leqslant2$.}
\step{综上所述，实数 $m$ 的取值范围是 $\{m\mid m\leqslant-4$ 或 $1\leqslant m\leqslant2\}$.}
\solend
\fi

\item 已知 $a>0$，关于 $x$ 的方程 $\dfrac1{x-2}+\dfrac1{x-1}=\dfrac{\sqrt{ax}}{x^2-3x+2}$ 有且仅有一个实数根，求实数 $a$ 的取值范围.

% 注：原卷【解析】按 Δ=0、Δ>0 分段（Δ=0 得 a=0 或 a=-24：a=-24 时 x=-3/2 舍去、a=0 时 x=3/2 符合），
%     结论印作 "综上，a≥0 且 a≠1/2"；该结论与题干 "a>0" 自相矛盾（a=0 时原方程分母含 √(ax) 而无意义）。
%     原卷如此，照录未改。
\ifanswer
\solbegin
\step{由题意得 $x\neq2,x\neq1$. 去分母整理得 $2x-3=\sqrt{ax}\geqslant0$，所以 $x\geqslant\dfrac32$，}
\step{故 $x\geqslant\dfrac32$ 且 $x\neq2$. 两边平方整理得 $4x^2-(12+a)x+9=0$，}
\step{判别式 $\Delta=a^2+24a$.}
\step{①$\Delta=0$，即 $a=0$ 或 $a=-24$.}
\step{当 $a=-24$ 时，$x=-\dfrac32$（舍去）；当 $a=0$ 时，$x=\dfrac32$ 符合题意.}
\step{②$\Delta>0$，即 $a>0$ 或 $a<-24$.}
\step{当 $a<-24$ 时，由韦达定理得两实数根为负，不符合题意，故舍去；}
\step{当 $a>0$ 时，由韦达定理得两实数根为正.}
\step{假设有一根为2，则 $a=\dfrac12$，此时另一根为 $\dfrac98$（$\dfrac98<\dfrac32$，故舍去）.}
\step{当方程在 $x>\dfrac32$ 时仅有一实根，则利用根的分布得 $f\left(\dfrac32\right)<0$，解得 $a>0$.}
\step{综上，$a\geqslant0$ 且 $a\neq\dfrac12$.}
\solend
\fi

\item 已知 $a\in\R$，关于 $x$ 的不等式 $ax^2-ax+3>0$ 恒成立.

\begin{enumerate}[label=(\arabic*)]
\item 求实数 $a$ 的取值范围；
\item 在（1）的条件下用反证法证明：三个关于 $x$ 的方程 $x^2+4ax-4a+3=0$，$x^2+(a-1)x+a^2=0$，$x^2+2ax-2a=0$ 中至少有一个方程有实数解.
\end{enumerate}

% 注：原卷(1)解析仅印“当 a=0 时，3>0 恒成立”与“当 a>0 时”两支，未讨论 a<0，
%     且 Δ 印作 a^2-4a×3、综上印作 0≤a<12；原卷如此，照录未改。
% 注：原卷(2)解析由判别式小于 0 的三支中只取 "a<-1" 一支推出 -3/2<a<-1 而与 (1) 矛盾，
%     "a>1/3" 一支（与 (1) 的范围不矛盾）未提；原卷如此，照录未改。
\ifanswer
\solbegin
\stepm{(1)}{当 $a=0$ 时，$3>0$ 恒成立；}
\step{当 $a>0$ 时，$\begin{cases}a>0\\\Delta=a^2-4a\times3<0\end{cases}$，解得 $0<a<12$；}
\step{综上，$0\leqslant a<12$.}
\stepm{(2)}{假设三个关于 $x$ 的方程 $x^2+4ax-4a+3=0$，$x^2+(a-1)x+a^2=0$，}
\step{$x^2+2ax-2a=0$ 都没有实数解，所以 $\begin{cases}16a^2-4(-4a+3)<0,\\(a-1)^2-4a^2<0,\\4a^2+4\times2a<0,\end{cases}$，}
\step{所以 $\begin{cases}a>\dfrac13\ \text{或}\ a<-1,\\-\dfrac32<a<\dfrac12,\\-2<a<0,\end{cases}$ 所以 $-\dfrac32<a<-1$，}
\step{与（1）的范围 $0\leqslant a<12$ 矛盾，所以假设不成立，}
\step{所以三个关于 $x$ 的方程 $x^2+4ax-4a+3=0$，$x^2+(a-1)x+a^2=0$，}
\step{$x^2+2ax-2a=0$ 中至少有一个方程有实数解.}
\solend
\fi

% 注：原卷条件③按字面未限定 2x∈P_n；若对所有 x∈P_n 都要求 2x∈A，原卷答案 f(4)=4 与之不符。
%     2012 年江苏卷源题将其表述为相对于 P_n 的补集条件；本稿仍照录高一42原文，未替换。
\item 设集合 $P_n=\{1,2,\cdots,n\}$，$n\in\N^*$. 记 $f(n)$ 为同时满足下列条件的集合 $A$ 的个数：

① $A\subseteq P_n$；②若 $x\in A$，则 $2x\notin A$；③若 $x\in\overline A$，则 $2x\notin\overline A$.

\begin{enumerate}[label=(\arabic*)]
\item 求 $f(4)$；
\item 求 $f(n)$ 的解析式（用 $n$ 表示）.
\end{enumerate}

\ifanswer
\solbegin
\stepm{(1)}{当 $n=4$ 时，$P_4=\{1,2,3,4\}$，}
\step{符合条件的集合 $A$ 为 $\{2\}$、$\{1,4\}$、$\{2,3\}$、$\{1,3,4\}$，故 $f(4)=4$；}
\stepm{(2)}{任取偶数 $x\in p_n$，将 $x$ 除以 2，若商仍为偶数，再除以 2$\cdots$，}
\step{经过 $k$ 次后，商必为奇数，此时记商为 $m$，}
\step{于是 $x=m\cdot2^k$，其中 $m$ 为奇数，$k\in\N^*$，}
\step{由条件得，若 $m\in A$，则 $x\in A\Longleftrightarrow k$ 为偶数，}
\step{若 $m\notin A$，则 $x\in A\Longleftrightarrow k$ 为奇数，}
\step{于是 $x$ 是否属于 $A$ 由 $m$ 是否属于 $A$ 确定，设 $Q_n$ 是 $P_n$ 中所有的奇数的集合，}
\step{因此 $f(n)$ 等于 $Q_n$ 的子集个数，}
\step{当 $n$ 为偶数时（或奇数时），$P_n$ 中奇数的个数是 $\dfrac{1}{2}n$（或 $\dfrac{1+n}{2}$），}
\step{所以 $f(n)=\begin{cases}2^{\frac n2},&n\text{ 为偶数},\\2^{\frac{n+1}{2}},&n\text{ 为奇数} .\end{cases}$}
\solend
\fi

\item 设 $S_n=\{(x_1,x_2,\cdots,x_n)\mid x_i\in\{0,1\},\ i=1,2,\cdots,n\}$（$n$ 为正整数），对任意的 $\alpha=(x_1,x_2,\cdots,x_n)$，$\beta=(y_1,y_2,\cdots,y_n)$，定义 $\alpha\cdot\beta=x_1y_1+x_2y_2+\cdots+x_ny_n$.

\begin{enumerate}[label=(\arabic*)]
\item 当 $n=3$ 时，$\alpha=(1,1,0)$，$\beta=(1,0,1)$，求 $\alpha\cdot\beta$；
\item 当 $n=3$ 时，集合 $A\subseteq S_n$，对于任意 $\alpha,\beta\in A$，$\alpha\cdot\beta$ 均为偶数，求 $A$ 中元素个数的最大值；
\item 集合 $A\subseteq S_n$，对于任意 $\alpha,\beta\in A$，$\alpha\neq\beta$，均有 $\alpha\cdot\beta\neq0$，求 $A$ 中元素个数的最大值.
\end{enumerate}

\ifanswer
\solbegin
\stepm{(1)}{当 $n=3$ 时，$\alpha\cdot\beta=x_1y_1+x_2y_2+x_3y_3=1\times1+1\times0+0\times1=1$；}
\stepm{(2)}{因为 $\alpha\cdot\beta=x_1y_1+x_2y_2+x_3y_3$ 均为偶数，所以结果为 $0$ 或 $2$，}
\step{若 $\alpha\cdot\beta=0$，则 $A$ 中的任意两个元素乘积为 $0$，}
\step{即 $(0,0,0),(0,0,1),(0,1,0),(1,0,0)$ 共有四个元素，}
\step{若 $\alpha\cdot\beta=2$，则 $A$ 中必有两个位置为 $1$，即 $(0,1,1),(1,1,1)$，}
\step{所以 $A$ 中元素个数的最大值为 $4$；}
\stepm{(3)}{$\alpha=(x_1,x_2,\cdots,x_n)$，$\alpha$ 中的“1”变为“0”，“0”变为“1”，}
\step{得到 $\gamma=(1-x_1,1-x_2,\cdots,1-x_n)$，得 $\alpha\cdot\gamma=0$，}
\step{因为 $\alpha\cdot\beta\neq0$，$\alpha\in A$，所以 $\gamma\notin A$，}
\step{因为 $S_n$ 中有 $2^n$ 个元素，所以 $A$ 中元素个数最多有 $\dfrac{2^n}{2}=2^{n-1}$ 个，}
\step{所以 $A$ 中元素个数的最大值为 $2^{n-1}$.}
\solend
