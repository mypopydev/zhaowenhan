\item 已知命题 $p$：“方程 $ax^2+2x+1=0$ 至少有一个负实根”，若 $p$ 为真命题的一个必要不充分条件为 $a\leqslant m+1$，则实数 $m$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
\step{若命题 $p$：“方程 $ax^2+2x+1=0$ 至少有一个负实根”为真命题，}
\step{当 $a=0$ 时，$2x+1=0$，$x=-\dfrac12$，符合题意；}
\step{当 $a<0$ 时，$\Delta=4-4a>0$，且 $x_1+x_2=-\dfrac2a>0$，$x_1x_2=\dfrac1a<0$，}
\step{此时方程 $ax^2+2x+1=0$ 有一个正根和一个负根，符合题意；}
\step{当 $a>0$ 时，由 $\Delta=4-4a=0$，解得 $a=1$，此时方程为 $x^2+2x+1=(x+1)^2=0$，}
\step{$x=-1$ 符合题意；}
\step{由 $\Delta=4-4a>0$ 解得 $0<a<1$，此时 $x_1+x_2=-\dfrac2a<0$，$x_1x_2=\dfrac1a>0$，}
\step{此时方程 $ax^2+2x+1=0$ 有两个负根，符合题意.}
\step{综上所述，$p$ 为真命题时，$a$ 的取值范围是 $(-\infty,1]$，}
\step{若 $p$ 为真命题的一个必要不充分条件为 $a\leqslant m+1$，}
\step{则 $m+1>1$，$m>0$，实数 $m$ 的取值范围是 $(0,+\infty)$.}
\solend

\fi

\item 已知全集 $U=\{a,b,c,d,e\}$，集合 $A$、$B$ 满足 $A\cap\overline{B}=\{a,b\}$，$\overline{A\cup B}=\{c,e\}$，则 $A=$\blankline[4.4em].

\ifanswer
\solbegin
\step{因为 $U=\{a,b,c,d,e\}$，$\overline{A\cup B}=\{c,e\}$，所以 $A\cup B=\{a,b,d\}$，}
\step{因为 $A\cap\overline B=\{a,b\}$，}
\step{所以 $A$ 中有 $a$、$b$ 这两个元素，$A$ 中没有 $c$、$e$ 这两个元素；}
\step{$B$ 中没有 $a$、$b$、$c$、$e$ 这四个元素，}
\step{若 $d\in A$，$d\in B$，则 $A=\{a,b,d\}$，$B=\{d\}$，所以 $\overline B=\{a,b,c,e\}$，}
\step{满足 $A\cup B=\{a,b,d\}$，也满足 $A\cap\overline B=\{a,b\}$，符合题意，则 $A=\{a,b,d\}$；}
\step{若 $d\in A$，$d\notin B$，则 $A=\{a,b,d\}$，$B=\varnothing$，所以 $\overline B=\{a,b,c,d,e\}$，}
\step{满足 $A\cup B=\{a,b,d\}$，不满足 $A\cap\overline B=\{a,b\}$，不符合题意；}
\step{若 $d\notin A$，$d\in B$，则 $A=\{a,b\}$，$B=\{d\}$，所以 $\overline B=\{a,b,c,e\}$，}
\step{满足 $A\cup B=\{a,b,d\}$，也满足 $A\cap\overline B=\{a,b\}$，符合题意，则 $A=\{a,b\}$；}
% 注：d∉A、d∉B 与 A∪B={a,b,d} 相矛盾，原卷仍照此讨论并推出 A={a,b}、B=∅，照录未改。
\step{若 $d\notin A$，$d\notin B$，则 $A=\{a,b\}$，$B=\varnothing$，所以 $\overline B=\{a,b,c,d,e\}$，}
\step{也满足 $A\cap\overline B=\{a,b\}$，不满足 $A\cup B=\{a,b,d\}$，不符合题意.}
\step{故 $A=\{a,b,d\}$ 或 $A=\{a,b\}$.}
\solend

\fi

\item 已知 $p$：$\dfrac{x-1}{x-3}\leqslant0$，$q$：$|x+2a|<2$，若 $p$ 是 $q$ 的充分不必要条件，则实数 $a$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
\step{$p:1\leqslant x<3$，$q:-2-2a<x<2-2a$，所以 $-2-2a<1$，$2-2a\geqslant3$，}
\step{所以 $-\dfrac32<a\leqslant-\dfrac12$.}
\solend

\fi

\item 已知 $a$、$b$、$c\in\R$，有四个推理：① $a>b\Rightarrow am^2>bm^2$；② $\dfrac ac>\dfrac bc\Rightarrow a>b$；③ $a>b$，$ab>0\Rightarrow\dfrac1a<\dfrac1b$；④ $a^2>b^2$，$ab>0\Rightarrow\dfrac1a<\dfrac1b$，其中正确的序号是 \blankline[4.4em].

\ifanswer
\solbegin
\step{对于①，当 $m=0$ 时，显然不等式不成立，故①错误；}
\step{对于②，当 $a=-3$，$b=-2$，$c=-1$ 时，$\dfrac ac>\dfrac bc$ 满足，$a>b$ 不满足，故②错误；}
\step{对于③，由 $ab>0$，则 $a\cdot\dfrac1{ab}>b\cdot\dfrac1{ab}$，即 $\dfrac1a<\dfrac1b$，故③正确；}
\step{对于④，由 $ab>0$ 得 $a$、$b$ 同号，故当 $a$、$b<0$ 时，$a^2>b^2$ 等价于 $a<b<0$，}
\step{故 $\dfrac1a>\dfrac1b$，故④错误.}
\step{故正确的是③.}
\solend

\fi

\item 关于 $x$ 的方程 $x^2+ax+a^2-8=0$ 有两个实根 $x_1$、$x_2$，且 $\dfrac1{x_1}+\dfrac1{x_2}=-\dfrac12$，则 $a$ 的值为 \blankline[4.4em].

\ifanswer
\solbegin
\step{由题意得 $\Delta=a^2-4(a^2-8)=32-3a^2\geqslant0$，解得 $a^2\leqslant\dfrac{32}{3}$，}
\step{由韦达定理得 $x_1+x_2=-a$，$x_1x_2=a^2-8$，则 $\dfrac1{x_1}+\dfrac1{x_2}=\dfrac{x_1+x_2}{x_1x_2}=\dfrac{-a}{a^2-8}=-\dfrac12$，}
\step{即 $a^2-2a-8=(a-4)(a+2)=0$，解得 $a=4$ 或 $a=-2$. 又 $a^2\leqslant\dfrac{32}{3}$，则 $a=-2$.}
\solend

\fi

\item 若不等式 $|x-a|+|x+2a|\geqslant -a+1$ 对于任意实数 $x$ 恒成立，则满足条件的实数 $a$ 的取值范围是 \blankline[4.4em].

\ifanswer
\solbegin
% 注：原卷此行印作 $|x-a-x-2a|$（未加括号，本稿原多加了括号），原卷如此，照录未改。
\step{由题意得 $|x-a|+|x+2a|\geqslant |x-a-x-2a|=3|a|\geqslant -a+1$，}
\step{所以 $\begin{cases}a\geqslant0\\3a\geqslant -a+1\end{cases}$ 或 $\begin{cases}a<0\\-3a\geqslant -a+1\end{cases}$，所以 $a\geqslant\dfrac14$ 或 $a\leqslant-\dfrac12$，}
\step{综上，满足条件的实数 $a$ 的取值范围是 $\left(-\infty,-\dfrac12\right]\cup\left[\dfrac14,+\infty\right)$.}
\solend

\fi

\item 已知正数 $a$、$b$、$c$ 满足 $c<1$，$a+b=4$，则 $\dfrac2{ab}+\dfrac1{bc(1-c)}$ 的最小值为 \blankline[4.4em].

\ifanswer
\solbegin
\step{由题意得 $c(1-c)\leqslant\left(\dfrac{c+1-c}{2}\right)^2=\dfrac14$，当 $c=\dfrac12$ 时取等号，}
\step{故 $\dfrac2{ab}+\dfrac1{bc(1-c)}\geqslant\dfrac2{ab}+\dfrac4b=\dfrac{a+b}{2ab}+\dfrac4b=\dfrac1{2a}+\dfrac9{2b}=\dfrac12\left(\dfrac1a+\dfrac9b\right)$}
\step{$=\dfrac18\left(\dfrac1a+\dfrac9b\right)(a+b)=\dfrac18\left(10+\dfrac ba+\dfrac{9a}b\right)\geqslant\dfrac18\left(10+2\sqrt{\dfrac ba\times\dfrac{9a}b}\right)=2$，}
\step{当 $b=3a=3$ 时取等号，}
\step{综上，当 $a=1$，$b=3$，$c=\dfrac12$ 时，$\dfrac2{ab}+\dfrac1{bc(1-c)}$ 的最小值为 $2$.}
\solend

\fi

\item 对于非空实数集合 $A$，记 $A^*=\{y\mid\text{对任意 }x\in A\text{，}y\leqslant x\}$，设非空实数集合 $P$ 满足条件“若 $x<1$，则 $x\notin P$”且 $M\subseteq P$，给出下列命题：

①若全集为实数 $\R$，对于任意非空实数集合 $A$，必有 $\overline A=A^*$；

②对于任意给定符合题设条件的集合 $M$、$P$，必有 $P^*\subseteq M^*$；

③存在符合题设条件的集合 $M$、$P$，使得 $M^*\cap P=\varnothing$；

④存在符合题设条件的集合 $M$、$P$，使得 $M\cap P^*\neq\varnothing$；

其中所有正确命题的序号是 \blankline[4.4em].

\ifanswer
\solbegin
\step{由于非空实数集 $A$，记 $A^*=\{y\mid\text{对任意 }x\in A\text{，}y\leqslant x\}$，}
\step{则 $A^*$ 中元素为不大于 $A$ 中所有值的数，即不大于 $A$ 中最小元素的集合；}
\step{①当 $A$ 集合下边界趋向负无穷大时，$A^*=\varnothing$，故①错误；}
% 注：题设的 M、P 为任意非空实数集合，未保证最大值存在；即使无最大值，P^*⊆M^* 仍由 M⊆P 直接得出，照录原解析的最大值论证。
\step{②由于 $M\subseteq P$，假设 $M$ 中最大值为 $m$，$P$ 最大值为 $p$，那么 $p\geqslant m$.}
\step{因此 $M^*$ 表示大于 $m$ 所有数集合，$P^*$ 表示所有大于 $p$ 的数的集合，}
\step{则 $P^*\subseteq M^*$，故②正确；}
\step{③令 $M=P=\{x\mid1<x<\dfrac32\}$，则 $M^*=\{y\mid1<x<\dfrac32\text{，且 }y\leqslant x\}=\{y\mid y<1\}$，}
\step{故 $M^*\cap P=\varnothing$，故③正确；}
% 注：下一例原卷取值含有小于 1 的元素，与题干“若 $x<1$，则 $x\notin P$”不符，照录未改。
\step{④令 $M=\{x\mid0\leqslant x<\dfrac12\}$，$P=\{x\mid0<x<\dfrac12\}$，}
\step{则 $P^*=\{x\mid x\leqslant0\}$，故 $M\cap P^*\neq\varnothing$，故④正确；}
\step{故所有正确命题的序号是②③④.}
\solend

\fi

\end{enumerate}

\item 已知命题 $p$：“$\exists x\in\R$，$ax^2+2x-1=0$”为假命题.

\begin{enumerate}[label=(\arabic*)]
\item 求实数 $a$ 的取值集合 $A$；
\item 设集合 $B=\{x\mid 6m-4<2x-4<2m\}$，若“$x\in A$”是“$x\in B$”的必要不充分条件，求实数 $m$ 的取值范围.
\end{enumerate}

\ifanswer
\solbegin
\stepm{(1)}{命题 $p:\exists x\in\R$，$ax^2+2x-1=0$ 为假命题，}
\step{即方程 $ax^2+2x-1=0$ 无实根，}
\step{若 $a=0$，则 $x=\dfrac12$，不符合题意；}
\step{若 $a\neq0$，则 $\Delta=4+4a<0$，解得 $a<-1$，}
\step{故实数 $a$ 的取值集合为 $A=\{a\mid a<-1\}$.}
\stepm{(2)}{集合 $B=\{x\mid3m<x<m+2\}$，}
\step{“$x\in A$”是“$x\in B$”的必要不充分条件，等价于集合 $B$ 是集合 $A$ 的真子集，}
\step{若 $B=\varnothing$，则 $3m\geqslant m+2$，即 $m\geqslant1$，此时集合 $B$ 是集合 $A$ 的真子集；}
\step{若 $B\neq\varnothing$，所以 $m+2>3m$，解得 $m<1$，}
\step{集合 $B$ 是集合 $A$ 的真子集，只要 $m+2\leqslant-1$，即 $m\leqslant-3$，}
\step{综上，实数 $m$ 的取值集合是 $\{m\mid m\leqslant-3\text{ 或 }m\geqslant1\}$.}
\solend

\fi

\item 果农种植一种水果，每年施肥和灌溉等需投入 $4$ 万元. 为提高产量同时改善水果口味以赢得市场，计划在今年投入 $x$ 万元用于改良品种. 根据其他果农种植经验发现，该水果年产量 $t$（万斤）与用于改良品种的资金投入 $x$（万元）之间的关系大致为：$t=3-\dfrac{m}{x+1}$（$x\geqslant0$，$m$ 为常数），若不改良品种，年产量为 $1$ 万斤. 该水果最初售价为每斤 $4$ 元，改良品种后，售价每斤提高 $\dfrac{x}{5}$ 元，假设产量和价格不受其他因素的影响.

\begin{enumerate}[label=(\arabic*)]
\item 求 $m$ 的值；
\item 设该果农种植该水果所获得的年利润为 $y$（万元）求当果农种植该水果所获得的利润不低于 $4$ 万元时，实数 $x$ 的取值范围；
% 注：原卷作"最大是利润"，系笔误（应为"最大利润"），照录未改。
\item 该果农一年内应当投入多少万元用于改良品种，才能使得年利润最大？最大是利润是多少？
\end{enumerate}

\ifanswer
\solbegin
\stepm{(1)}{若不改良品种，年产量为 $1$ 万斤，则 $x=0$ 时，$t=1$，即 $1=3-\dfrac m1$，解得 $m=2$；}
% 注：本稿原把“万斤”误作“万吨”，并据此误记“未作单位换算”的存疑；2026-10-05 已照原卷改回“万斤”（万斤 × 元/斤 即万元，无单位换算问题）。
\stepm{(2)}{因为年产量 $t=3-\dfrac2{x+1}$ 万斤，改良后售价 $p=\left(4+\dfrac x5\right)$ 元/斤，}
\step{总成本 $4+x$ 万元，}
\step{所以利润 $y=\left(3-\dfrac2{x+1}\right)\left(4+\dfrac x5\right)-(4+x)=12+\dfrac{3x}{5}-\dfrac8{x+1}-\dfrac{2x}{5(x+1)}-4-x$}
\step{$=8-\dfrac{2x}{5}-\dfrac8{x+1}-\dfrac{2x}{5(x+1)}=\dfrac{2x(18-x)}{5(x+1)}$（$x\geqslant0$），}
\step{若利润不低于 $4$ 万元，则 $y=\dfrac{2x(18-x)}{5(x+1)}\geqslant4$，}
\step{因为 $5(x+1)\geqslant0$，所以 $2x(18-x)\geqslant20(x+1)$，}
\step{化简并整理得 $x^2-8x+10\leqslant0$，解得 $4-\sqrt6\leqslant x\leqslant4+\sqrt6$，}
\step{所以 $x\in[4-\sqrt6,4+\sqrt6]$；}
\stepm{(3)}{因为 $y=8-\dfrac{2x}{5}-\dfrac8{x+1}-\dfrac{2x}{5(x+1)}=\dfrac{2x(18-x)}{5(x+1)}$，}
\step{所以 $y=8-\left[\dfrac{2(x+1)}5+\dfrac{38}{5(x+1)}\right]$，}
\step{$\leqslant8-2\sqrt{\dfrac{2(x+1)}5\cdot\dfrac{38}{5(x+1)}}=8-\dfrac{4\sqrt{19}}5$，}
\step{当且仅当 $\dfrac{2(x+1)}5=\dfrac{38}{5(x+1)}$，即 $x=\sqrt{19}-1$ 时取等号，}
\step{所以果农一年内应当投入 $x=\sqrt{19}-1$ 万元用于改良品种，获得最大利润，}
\step{最大利润为 $8-\dfrac{4\sqrt{19}}5$ 万元.}
\solend

\fi

\item 设函数 $f(x)=ax^2+(b-2)x+c+3$（$a\neq0$）.

\begin{enumerate}[label=(\arabic*)]
\item 当 $c=0$ 时，不等式 $f(x)<0$ 的解集为 $(-3,-1)$，求实数 $a$、$b$ 的值；
\item 若 $b=-a-1$，求关于 $x$ 的不等式 $f(x)>-4x+c+4$ 的解集；
\item 若 $f(x)\geqslant(2a-2)x+b+3$ 对 $\forall x\in\R$ 恒成立，求 $\dfrac{b^2}{3a^2+c^2}$ 的最大值.
\end{enumerate}

\ifanswer
\solbegin
\stepm{(1)}{当 $c=0$ 时，$f(x)=ax^2+(b-2)x+3$，$f(x)<0$ 的解集为 $(-3,-1)$，}
\step{故方程 $ax^2+(b-2)x+3=0$ 的根为 $-3$、$-1$，且 $a>0$；}
\step{由韦达定理得 $(-3)\times(-1)=\dfrac3a$，解得 $a=1$；}
\step{$-3+(-1)=-\dfrac{b-2}{a}$，代入 $a=1$ 得 $-4=-(b-2)$，解得 $b=6$.}
\stepm{(2)}{将 $b=-a-1$ 代入不等式 $f(x)>-4x+c+4$，}
\step{整理得 $ax^2+(1-a)x-1>0$，因式分解为 $(ax+1)(x-1)>0$.}
\step{当 $a>0$ 时，方程 $(ax+1)(x-1)=0$ 的根为 $x=-\dfrac1a$（小于 $1$），}
\step{解集为 $\left(-\infty,-\dfrac1a\right)\cup(1,+\infty)$；}
\step{当 $a=-1$ 时，不等式化为 $-(x-1)^2>0$，无实数解，解集为 $\varnothing$；}
\step{当 $a<-1$ 时，$-\dfrac1a<1$，不等式解集为 $\left(-\dfrac1a,1\right)$；}
\step{当 $-1<a<0$ 时，$-\dfrac1a>1$，不等式解集为 $\left(1,-\dfrac1a\right)$.}
\stepm{(3)}{整理不等式 $f(x)\geqslant(2a-2)x+b+3$ 得 $ax^2+(b-2a)x-b+c\geqslant0$，}
\step{对 $\forall x\in\R$ 恒成立，故 $a>0$ 且判别式 $\Delta=(b-2a)^2-4a(-b+c)\leqslant0$，}
\step{化简得 $4ac\geqslant b^2+4a^2$，即 $c\geqslant\dfrac{b^2+4a^2}{4a}$.}
\step{令 $t=\dfrac ba$（$t\in\R$），则 $b=ta$，代入得 $c\geqslant\dfrac{t^2a^2+4a^2}{4a}=\dfrac{a(t^2+4)}4$.}
\step{将 $b=ta$，$c\geqslant\dfrac{a(t^2+4)}4$ 代入 $\dfrac{b^2}{3a^2+c^2}$，}
\step{得 $\dfrac{b^2}{3a^2+c^2}\leqslant\dfrac{t^2a^2}{3a^2+\dfrac{a^2(t^2+4)^2}{16}}=\dfrac{16t^2}{48+(t^2+4)^2}$，}
\step{令 $u=t^2\geqslant0$，表达式化为 $\dfrac{16u}{u^2+8u+64}$，}
\step{分子分母同除以 $u$（$u>0$）得 $\dfrac{16}{u+\dfrac{64}{u}+8}$，}
\step{由基本不等式 $u+\dfrac{64}{u}\geqslant2\sqrt{u\times\dfrac{64}{u}}=16$（当且仅当 $u=8$ 时取等号），}
\step{故 $\dfrac{16}{u+\dfrac{64}{u}+8}\leqslant\dfrac{16}{16+8}=\dfrac23$.}
\solend

\end{enumerate}
